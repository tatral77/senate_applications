	   BEGIN

		SELECT        GROUPS.DESCRIPTION AS GROUPDESC, GROUPS.ASSEMBLY_SESSIONS_ID, GROUP_DIVISIONS.GROUP_DIVISIONS_ID,GROUP_DIVISIONS.GROUPS_ID, GROUP_DIVISIONS.DIVISIONS_ID, DIVISIONS.DESCRIPTION AS DIVISION, DIVISIONS.IS_ACTIVE, 
                      GROUPS.CORRESPONDING_GROUP, GROUPS.GROUP_SEQUENCE
		FROM          GROUPS INNER JOIN
                      GROUP_DIVISIONS ON GROUPS.GROUPS_ID = GROUP_DIVISIONS.GROUPS_ID INNER JOIN
                      DIVISIONS ON GROUP_DIVISIONS.DIVISIONS_ID = DIVISIONS.DIVISIONS_ID
		WHERE        (GROUPS.ASSEMBLY_SESSIONS_ID =   @AssemblySessionsId) AND GROUP_SEQUENCE IN(1,2,3,4,5)
		ORDER BY GROUPS.CORRESPONDING_GROUP, DIVISION
