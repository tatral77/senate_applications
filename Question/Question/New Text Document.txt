  IF @Action = 'GET_DIVISIONS'
	   BEGIN
	   SELECT        DIVISIONS.DIVISIONS_ID, DIVISIONS.DESCRIPTION AS DIVISION, DIVISIONS.MINISTRIES_ID, DIVISIONS.IS_ACTIVE, MINISTRIES.DESCRIPTION AS MINISTRY
		FROM            DIVISIONS INNER JOIN
                         MINISTRIES ON DIVISIONS.MINISTRIES_ID = MINISTRIES.MINISTRIES_ID 
						 ORDER BY DIVISIONS.DESCRIPTION
	   END

	   IF @Action = 'INSERT_DIVISIONS'
	   BEGIN
	    Insert into DIVISIONS(DESCRIPTION, MINISTRIES_ID, IS_ACTIVE)
		VALUES ( @Description, @MinistriesId, @IsActive)
   		END

		IF @Action = 'UPDATE_DIVISIONS'
	   BEGIN
	   update DIVISIONS set
	   DESCRIPTION=@Description,
	   MINISTRIES_ID=@MinistriesId,
	   IS_ACTIVE=@IsActive
	    where
	   DIVISIONS_ID=@DivisionId
   	  END
