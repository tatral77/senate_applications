BEGIN

	   IF @Action = 'GET_MOTION_MOVERS'
	   BEGIN

FROM            SYSTEM_USERS INNER JOIN
                         MB_MOTION_MOVERS ON SYSTEM_USERS.SYSTEM_USERS_ID = MB_MOTION_MOVERS.SYSTEM_USERS_ID INNER JOIN
                         MB_MOTIONS ON MB_MOTION_MOVERS.MOTIONS_ID = MB_MOTIONS.MOTIONS_ID

	    IF @Action = 'GET_MOTION_MOVER'

	   SELECT       SYSTEM_USERS.F_NAME, MB_MOTION_MOVERS.MOTION_MOVERS_ID, MB_MOTION_MOVERS.MOTIONS_ID, MB_MOTION_MOVERS.SYSTEM_USERS_ID, MB_MOTIONS.VERSION_NO
FROM            SYSTEM_USERS INNER JOIN
                         MB_MOTION_MOVERS ON SYSTEM_USERS.SYSTEM_USERS_ID = MB_MOTION_MOVERS.SYSTEM_USERS_ID INNER JOIN
                         MB_MOTIONS ON MB_MOTION_MOVERS.MOTIONS_ID = MB_MOTIONS.MOTIONS_ID
						 WHERE MB_MOTION_MOVERS.MOTIONS_ID=@MotionsId AND MOTION_MOVERS_ID=@MotionMoversId;
	   END

	   IF @Action = 'INSERT_MOTION_MOVER'
	   BEGIN
	   Insert into MB_MOTION_MOVERS( MOTIONS_ID, SYSTEM_USERS_ID)
	   VALUES ( @MotionsId, @SystemUsersId)

	   END
	    IF @Action = 'DELETE_MOTION_MOVER'

FROM            MOTION_TYPES WHERE IS_FOR_MOTIONS=1;

FROM            MB_MOTIONS INNER JOIN
                         MOTION_STATUSES ON MB_MOTIONS.MOTION_STATUS = MOTION_STATUSES.MOTION_STATUSES_ID INNER JOIN
                         MOTION_TYPES ON MB_MOTIONS.MOTION_TYPE = MOTION_TYPES.MOTION_TYPES_ID INNER JOIN
                         ASSEMBLY_SESSIONS ON MB_MOTIONS.SESSION_NUMBER = ASSEMBLY_SESSIONS.ASSEMBLY_SESSIONS_ID INNER JOIN
                         MB_WEEKS ON MB_MOTIONS.MB_WEEK_ID = MB_WEEKS.MB_WEEKS_ID
			WHERE	     MB_MOTIONS.SESSION_NUMBER=@SessionNumber AND PROCESSED=1
	   END

			BEGIN
			SELECT        MB_MOTIONS.MOTIONS_ID, MB_MOTIONS.FILE_NO, MB_MOTIONS.MB_WEEK_ID, MB_MOTIONS.NOTICE_DIARY_NO, MB_MOTIONS.NOTICE_DIARY_DATE, MB_MOTIONS.NOTICE_DIARY_TIME, 
                         MB_MOTIONS.SESSION_NUMBER, MB_MOTIONS.REFFERED_SC_DATE, MB_MOTIONS.MOVING_HOUSE_DATE, MB_MOTIONS.DISCUSSION_DATE, MB_MOTIONS.DESCRIPTION AS MOTION_TEXT,SUBJECT, 
                         ASSEMBLY_SESSIONS.SESSION_NUMBER AS SESSION_DESC, MOTION_TYPES.DESCRIPTION AS MOTION_TYPE, MOTION_STATUSES.DESCRIPTION AS MOTION_STATUS, 
                         MB_MOTIONS.MOTION_TYPE AS MOTION_TYPE_ID, MB_MOTIONS.MOTION_DIARY_NO, MB_MOTIONS.MOTION_DIARY_DATE, MB_MOTIONS.MOTION_DIARY_TIME, MB_MOTIONS.MOTION_STATUS AS MOTION_STATUS_ID, 
                         MB_MOTIONS.ADMISSION_DATE, MB_MOTIONS.PRESENT_HOUSE_DATE, MB_WEEKS.MB_WEEK
FROM            MB_MOTIONS INNER JOIN
                         MOTION_STATUSES ON MB_MOTIONS.MOTION_STATUS = MOTION_STATUSES.MOTION_STATUSES_ID INNER JOIN
                         MOTION_TYPES ON MB_MOTIONS.MOTION_TYPE = MOTION_TYPES.MOTION_TYPES_ID INNER JOIN
                         ASSEMBLY_SESSIONS ON MB_MOTIONS.SESSION_NUMBER = ASSEMBLY_SESSIONS.ASSEMBLY_SESSIONS_ID INNER JOIN
                         MB_WEEKS ON MB_MOTIONS.MB_WEEK_ID = MB_WEEKS.MB_WEEKS_ID
WHERE        (MB_MOTIONS.MOTIONS_ID = @MotionsId);
END

			Insert into MB_MOTIONS(MOTIONS_ID,FILE_NO, NOTICE_DIARY_NO, NOTICE_DIARY_DATE,NOTICE_DIARY_TIME, SESSION_NUMBER,MOTION_DIARY_NO,MOTION_TYPE,MOTION_STATUS,MB_WEEK_ID,IS_ACTIVE,SUBJECT,PROCESSED )
			VALUES ( @MotionsId,@FileNo,@NoticeDiaryNo, @NoticeDiaryDate,@NoticeDiaryTime ,@SessionNumber,@MotionDiaryNo,@MotionType,@MotionStatusesId,@WeekId,1,@Subject,1) 
		
	   END

		   MOVING_HOUSE_DATE=@MovingHouseDate,
		   DISCUSSION_DATE=@DiscussionDate,
		   SUBJECT=@Subject,
		   PROCESSED=1
		    where 
			MOTIONS_ID=@MotionsId

		END
		IF @Action = 'DELETE_MB_MOTION'
			BEGIN
			Delete from MB_MOTION_MOVERS where MB_MOTION_MOVERS.MOTIONS_ID=@MotionsId;
			Delete from MB_MOTION_MINISTRIES WHERE MB_MOTION_MINISTRIES.MOTION_ID=@MotionsId;
			Delete from MB_MOTIONS where MOTIONS_ID=@MotionsId;
		END

	   BEGIN
	   SELECT        MB_WEEKS_ID, MB_WEEK

IF @Action = 'STATUS_NO_OF_MOTIONS'
BEGIN
SELECT        MB_MOTIONS.MOTION_STATUS, ASSEMBLY_SESSIONS.SESSION_NUMBER, MOTION_STATUSES.DESCRIPTION, MOTION_STATUSES.MOTION_STATUSES_ID
FROM            MB_MOTIONS INNER JOIN
                         ASSEMBLY_SESSIONS ON MB_MOTIONS.SESSION_NUMBER = ASSEMBLY_SESSIONS.ASSEMBLY_SESSIONS_ID INNER JOIN
                         MOTION_STATUSES ON MB_MOTIONS.MOTION_STATUS =MOTION_STATUSES.MOTION_STATUSES_ID
WHERE     ((MB_MOTIONS.SESSION_NUMBER >= @FromSession) AND (MB_MOTIONS.SESSION_NUMBER <= @ToSession)  AND (MB_MOTIONS.MOTION_TYPE = @MotionTypesId))     
			AND MB_MOTIONS.MOTION_STATUS IN
     (SELECT DISTINCT MOTION_STATUS FROM MB_MOTIONS WHERE (MB_MOTIONS.SESSION_NUMBER >= @FromSession) AND (MB_MOTIONS.SESSION_NUMBER <= @ToSession))
        ORDER BY MOTION_STATUSES.DESCRIPTION

END
